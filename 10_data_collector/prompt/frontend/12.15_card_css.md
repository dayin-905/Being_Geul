function init3DEffect() {

    // [핵심 기능] 화면 너비에 따라 적절한 확대 배율과 Y축 이동 거리를 반환하는 함수
    // Tailwind의 Breakpoint(sm, lg, xl)와 논리를 맞췄습니다.
    const getResponsiveSettings = () => {
        const width = window.innerWidth;
        
        if (width >= 1280) { // xl (데스크탑 - 4열)
            return { scale: 1.2, y: -15 }; 
        } else if (width >= 1024) { // lg (노트북 - 3열)
            return { scale: 1.15, y: -12 };
        } else if (width >= 640) { // sm (태블릿 - 2열)
            return { scale: 1.08, y: -8 };
        } else { // 기본 (모바일 - 1열)
            return { scale: 1.03, y: -4 };
        }
    };

    wrappers.forEach(wrapper => {
        const card = wrapper.querySelector('.policy-card');
        const glare = wrapper.querySelector('.card-glare');

        // --- 1. 진입 효과 (Hover & Focus) ---
        const handleEnter = () => {
            // [수정] 함수를 호출해 현재 화면 크기에 맞는 값을 가져옴
            const settings = getResponsiveSettings();

            gsap.set(wrapper, { zIndex: 50 });

            gsap.to(card, {
                duration: 0.4,
                scale: settings.scale,  // 받아온 스케일 값 적용
                y: settings.y,          // 받아온 Y축 이동 값 적용
                boxShadow: "0 30px 60px -10px rgba(0,0,0,0.25)",
                ease: "back.out(1.5)",
                overwrite: "auto"
            });
        };

        wrapper.addEventListener('mouseenter', handleEnter);
        card.addEventListener('focusin', handleEnter);


        // --- 2. 움직임 효과 (Tilt) ---
        wrapper.addEventListener('mousemove', (e) => {
            // 모바일(< 768px)에서는 기울기 효과 끄기 (성능 및 UX 보호)
            if (window.innerWidth < 768) return;

            const rect = wrapper.getBoundingClientRect();
            // 마우스 위치 계산
            const x = e.clientX - rect.left;
            const y = e.clientY - rect.top;
            const centerX = rect.width / 2;
            const centerY = rect.height / 2;

            // 각도 계산 (너무 과하지 않게 10도로 제한)
            const rotateX = ((y - centerY) / centerY) * -10;
            const rotateY = ((x - centerX) / centerX) * 10;

            gsap.to(card, {
                duration: 0.2,
                rotateX: rotateX,
                rotateY: rotateY,
                transformPerspective: 1000,
                ease: "power1.out",
                overwrite: "auto"
            });

            if (glare) {
                gsap.to(glare, {
                    duration: 0.2,
                    opacity: 0.6,
                    backgroundPosition: `${(x / rect.width) * 100}% ${(y / rect.height) * 100}%`
                });
            }
        });


        // --- 3. 나감 효과 (Leave & Blur) ---
        const handleLeave = () => {
            gsap.delayedCall(0.3, () => { gsap.set(wrapper, { zIndex: 1 }); });

            gsap.to(card, {
                duration: 0.6,
                scale: 1,
                y: 0,
                rotateX: 0,
                rotateY: 0,
                boxShadow: "0 1px 2px 0 rgba(0, 0, 0, 0.05)",
                ease: "power3.out",
                overwrite: true
            });

            if (glare) {
                gsap.to(glare, { duration: 0.5, opacity: 0 });
            }
        };

        card.addEventListener('focusout', handleLeave);
    });

    gsap.from(".perspective-wrap", {

        opacity: 0,
        duration: 0.8,
        stagger: 0.1,
        ease: "power2.out"
    });
